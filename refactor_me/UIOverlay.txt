import React from 'react';
import { Activity, Sparkles, Sword, Zap } from 'lucide-react';
import { GameController } from '../game/GameController';

interface UIOverlayProps {
  controller: GameController;
  themeColor: string;
  setThemeColor: (color: string) => void;
  combo: number;
  score: number;
  maxCombo: number;
}

const THEMES = [
  { name: 'Neon Cyan', color: '#06b6d4' },
  { name: 'Acid Green', color: '#22c55e' },
  { name: 'Hot Crimson', color: '#f43f5e' },
  { name: 'Hyper Violet', color: '#a855f7' },
  { name: 'Molten Gold', color: '#eab308' },
];

export const UIOverlay: React.FC<UIOverlayProps> = ({
  controller,
  themeColor,
  setThemeColor,
  combo,
  score,
  maxCombo,
}) => {
  // Use local state trigger to force UI refresh on user clicks
  const [, setRefresh] = React.useState(0);

  const handleSimulatedKeyPress = (code: string) => {
    controller.handleKeyDown(code, code);
    setTimeout(() => {
      controller.handleKeyUp(code, code);
    }, 150);
    setRefresh(r => r + 1);
  };

  return (
    <div className="absolute inset-0 pointer-events-none flex flex-col justify-between p-6 select-none font-sans">
      {/* Top Bar: Title, Stats, and Instructions */}
      <div className="flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
        {/* Header Title */}
        <div className="pointer-events-auto bg-slate-950/80 backdrop-blur border border-slate-800 p-4 rounded-xl shadow-2xl">
          <div className="flex items-center gap-2">
            <Activity className="w-6 h-6 animate-pulse" style={{ color: themeColor }} />
            <h1 className="text-xl font-black tracking-wider text-white">
              SILHOUETTE <span style={{ color: themeColor }}>BALLET</span>
            </h1>
          </div>
          <p className="text-xs text-slate-400 mt-1 max-w-xs">
            Single Contour Line Upper-Envelope Platformer. Ground and entities are fused using Signed Distance Fields.
          </p>
        </div>

        {/* Live Metrics */}
        <div className="flex items-center gap-4 pointer-events-auto">
          <div className="bg-slate-950/80 backdrop-blur border border-slate-800 px-4 py-2 rounded-xl text-center">
            <div className="text-[10px] text-slate-400 uppercase font-bold tracking-wider">Score</div>
            <div className="text-lg font-black text-white">{score.toLocaleString()}</div>
          </div>
          <div className="bg-slate-950/80 backdrop-blur border border-slate-800 px-4 py-2 rounded-xl text-center">
            <div className="text-[10px] text-slate-400 uppercase font-bold tracking-wider">Max Combo</div>
            <div className="text-lg font-black" style={{ color: themeColor }}>{maxCombo}x</div>
          </div>
        </div>
      </div>

      {/* Massive Dopamine Combo Splash Center-Top */}
      <div className="flex-1 flex items-start justify-center mt-8 pointer-events-none">
        {combo > 0 && (
          <div className="animate-bounce transition-all duration-75 text-center">
            <div 
              className="text-7xl font-black tracking-tighter drop-shadow-[0_10px_20px_rgba(0,0,0,0.8)]"
              style={{ color: themeColor, textShadow: `0 0 30px ${themeColor}` }}
            >
              {combo}<span className="text-4xl text-white">x</span>
            </div>
            <div className="text-sm font-extrabold text-white tracking-widest uppercase bg-slate-900/90 px-4 py-1 rounded-full border border-slate-700/50 inline-block mt-2 animate-pulse">
              <Sparkles className="w-4 h-4 inline mr-1 pb-0.5 text-yellow-400" />
              BALLET OF BRUTALITY
            </div>
          </div>
        )}
      </div>

      {/* Bottom Interface: Customizer, Controls Sheet, Attack Hotkeys */}
      <div className="flex flex-col md:flex-row justify-between items-end gap-6">
        {/* Left: Toggles & Color Themes */}
        <div className="pointer-events-auto flex flex-col gap-3 bg-slate-950/80 backdrop-blur border border-slate-800 p-4 rounded-xl max-w-sm">
          <div className="text-xs font-bold text-slate-300 flex items-center gap-2 pb-1 border-b border-slate-800">
            <span>SDF Contour Customizer</span>
          </div>
          
          {/* Color buttons */}
          <div className="flex flex-wrap gap-2">
            {THEMES.map(t => (
              <button
                key={t.name}
                onClick={() => setThemeColor(t.color)}
                className={`w-6 h-6 rounded-full border transition-all hover:scale-110 ${
                  themeColor === t.color ? 'ring-2 ring-white scale-110' : 'opacity-60'
                }`}
                style={{ backgroundColor: t.color }}
                title={t.name}
              />
            ))}
          </div>

          {/* Dev Mode toggle */}
          <label className="flex items-center gap-2 cursor-pointer mt-1 pt-2 border-t border-slate-800/80">
            <input
              type="checkbox"
              checked={controller.devMode}
              onChange={e => {
                controller.devMode = e.target.checked;
                setRefresh(r => r + 1);
              }}
              className="rounded bg-slate-900 border-slate-700 text-cyan-500 focus:ring-0"
            />
            <span className="text-xs font-semibold text-slate-300">
              Dev Mode (View Underlying Stick Skeletons)
            </span>
          </label>
        </div>

        {/* Center/Right Controls Cheat-Sheet & Fast Action Triggers */}
        <div className="pointer-events-auto flex flex-col items-end gap-3">
          {/* Key bindings hint */}
          <div className="hidden sm:flex items-center gap-2 bg-slate-950/80 backdrop-blur border border-slate-800 px-4 py-2 rounded-xl text-xs text-slate-400">
            <span>Move: <strong className="text-white">A / D</strong> or <strong className="text-white">Arrows</strong></span>
            <span className="text-slate-600">|</span>
            <span>Jump: <strong className="text-white">W / Space</strong></span>
            <span className="text-slate-600">|</span>
            <span>Attacks: <strong className="text-white">1, 2, 3</strong></span>
          </div>

          {/* Graphical Attack Trigger Buttons for convenience/mobile mashing */}
          <div className="flex gap-2">
            <button
              onClick={() => handleSimulatedKeyPress('Digit1')}
              className="flex flex-col items-center justify-center w-20 h-16 bg-slate-900 hover:bg-slate-800 active:scale-95 border-2 rounded-xl transition-all shadow-lg text-white"
              style={{ borderColor: `${themeColor}60` }}
            >
              <Sword className="w-5 h-5 text-slate-300" />
              <span className="text-[10px] font-bold mt-1">1: Sweep</span>
            </button>
            
            <button
              onClick={() => handleSimulatedKeyPress('Digit2')}
              className="flex flex-col items-center justify-center w-20 h-16 bg-slate-900 hover:bg-slate-800 active:scale-95 border-2 rounded-xl transition-all shadow-lg text-white"
              style={{ borderColor: themeColor }}
            >
              <Zap className="w-5 h-5 animate-pulse" style={{ color: themeColor }} />
              <span className="text-[10px] font-bold mt-1">2: Roundhouse</span>
            </button>

            <button
              onClick={() => handleSimulatedKeyPress('Digit3')}
              className="flex flex-col items-center justify-center w-20 h-16 bg-pink-950 hover:bg-pink-900 active:scale-95 border-2 border-pink-500 rounded-xl transition-all shadow-lg text-white"
            >
              <Sparkles className="w-5 h-5 text-pink-400" />
              <span className="text-[10px] font-bold mt-1">3: Slam</span>
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};
