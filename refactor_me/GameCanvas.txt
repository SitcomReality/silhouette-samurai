import React, { useEffect, useRef } from 'react';
import { GameController } from '../game/GameController';
import { computeContourY, extractSDFSegments } from '../game/SDFMath';
import { Skeleton } from '../game/types';

interface GameCanvasProps {
  controller: GameController;
  themeColor: string;
}

export const GameCanvas: React.FC<GameCanvasProps> = ({ controller, themeColor }) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let animationId: number;
    let lastTime = performance.now();

    const handleResize = () => {
      if (!canvas.parentElement) return;
      canvas.width = canvas.parentElement.clientWidth;
      canvas.height = canvas.parentElement.clientHeight;
    };
    handleResize();
    window.addEventListener('resize', handleResize);

    const render = (time: number) => {
      const dt = Math.min((time - lastTime) / 1000, 0.1);
      lastTime = time;

      // Update Simulation Engine
      controller.update(dt);

      // Clear Canvas
      ctx.save();
      ctx.fillStyle = '#090d16'; // sleek deep dark void
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      // Apply Juicy Visceral Screen Shake
      if (controller.screenShake > 0) {
        const shakeX = (Math.random() - 0.5) * controller.screenShake;
        const shakeY = (Math.random() - 0.5) * controller.screenShake;
        ctx.translate(shakeX, shakeY);
      }

      const viewWidth = canvas.width;
      const viewHeight = canvas.height;
      const camYOffset = viewHeight * 0.65 - controller.baseGroundY; 

      // 1. FAST GROUND FILL SCAN (Translucent gradient beneath grounded terrain)
      const groundedEntities = controller.entities.concat(controller.player).filter(e => e.isGrounded);
      const groundedParticles = controller.particles.filter(p => p.isGrounded);

      const columns = Math.ceil(viewWidth / 2.0); // optimized scan step for fill region
      const stepPixels = viewWidth / columns;
      
      ctx.beginPath();
      let firstY = 0;
      for (let i = 0; i <= columns; i++) {
        const cx = i * stepPixels;
        const worldX = controller.cameraX + (cx - viewWidth / 2);
        const worldY = computeContourY(
          worldX,
          controller.baseGroundY,
          groundedEntities,
          groundedParticles,
          controller.shockwaves,
          20
        );
        const cy = worldY + camYOffset;
        if (i === 0) {
          firstY = cy;
          ctx.moveTo(cx, cy);
        } else {
          ctx.lineTo(cx, cy);
        }
      }
      ctx.lineTo(viewWidth, viewHeight);
      ctx.lineTo(0, viewHeight);
      ctx.lineTo(0, firstY);
      ctx.closePath();

      const gradient = ctx.createLinearGradient(0, viewHeight * 0.4, 0, viewHeight);
      gradient.addColorStop(0, `${themeColor}20`); // subtle tint beneath surface
      gradient.addColorStop(1, `${themeColor}02`);
      ctx.fillStyle = gradient;
      ctx.fill();

      // 2. UNIFIED GLOBAL SDF MARCHING SQUARES MONOLINE STROKE
      // Extracts global contiguous shared isolines encompassing terrain, airborne characters, and particles.
      // Positively prevents outline overlap, guarantees absolutely identical aesthetics, and natively pinches shapes.
      const allEntities = controller.entities.concat(controller.player);
      const segments = extractSDFSegments(
        viewWidth,
        viewHeight,
        controller.cameraX,
        camYOffset,
        controller.baseGroundY,
        allEntities,
        controller.particles,
        controller.shockwaves,
        4.0, // High-fidelity 4px grid step gives beautiful, buttery smooth sub-pixel curves
        20
      );

      ctx.beginPath();
      for (let i = 0; i < segments.length; i++) {
        const seg = segments[i];
        ctx.moveTo(seg.x1, seg.y1);
        ctx.lineTo(seg.x2, seg.y2);
      }

      // Configure round linecaps/joins to perfectly fuse grid segments into continuous vectors
      ctx.lineCap = 'round';
      ctx.lineJoin = 'round';

      // Outer vibrant colored luminous pass
      ctx.lineWidth = 4;
      ctx.strokeStyle = themeColor;
      ctx.shadowColor = themeColor;
      ctx.shadowBlur = 12;
      ctx.stroke();

      // Inner brilliant core pass for perfect neon monoline definition
      ctx.lineWidth = 2;
      ctx.strokeStyle = '#ffffff';
      ctx.shadowBlur = 3;
      ctx.stroke();
      
      ctx.shadowBlur = 0; // reset shadow

      // 3. Dev Debug Overlay: Skeletons & Shapes
      if (controller.devMode) {
        const drawSkeletonDebug = (skel: Skeleton, color: string) => {
          const mapP = (p: {x: number, y: number}) => ({
            x: (p.x - controller.cameraX) + viewWidth / 2,
            y: p.y + camYOffset
          });

          ctx.lineWidth = 1.5;
          ctx.strokeStyle = color;
          ctx.fillStyle = color;

          const h = mapP(skel.head);
          ctx.beginPath();
          ctx.arc(h.x, h.y, skel.headRadius, 0, Math.PI * 2);
          ctx.stroke();

          const tt = mapP(skel.torsoTop);
          const tb = mapP(skel.torsoBottom);
          ctx.beginPath();
          ctx.moveTo(tt.x, tt.y); ctx.lineTo(tb.x, tb.y);
          ctx.stroke();

          const ls = mapP(skel.leftShoulder);
          const le = mapP(skel.leftElbow);
          const lh = mapP(skel.leftHand);
          ctx.beginPath(); ctx.moveTo(ls.x, ls.y); ctx.lineTo(le.x, le.y); ctx.lineTo(lh.x, lh.y); ctx.stroke();

          const rs = mapP(skel.rightShoulder);
          const re = mapP(skel.rightElbow);
          const rh = mapP(skel.rightHand);
          ctx.beginPath(); ctx.moveTo(rs.x, rs.y); ctx.lineTo(re.x, re.y); ctx.lineTo(rh.x, rh.y); ctx.stroke();

          const lhip = mapP(skel.leftHip);
          const lk = mapP(skel.leftKnee);
          const lf = mapP(skel.leftFoot);
          ctx.beginPath(); ctx.moveTo(lhip.x, lhip.y); ctx.lineTo(lk.x, lk.y); ctx.lineTo(lf.x, lf.y); ctx.stroke();

          const rhip = mapP(skel.rightHip);
          const rk = mapP(skel.rightKnee);
          const rf = mapP(skel.rightFoot);
          ctx.beginPath(); ctx.moveTo(rhip.x, rhip.y); ctx.lineTo(rk.x, rk.y); ctx.lineTo(rf.x, rf.y); ctx.stroke();

          [tt, tb, le, re, lk, rk].forEach(pt => {
            ctx.beginPath(); ctx.arc(pt.x, pt.y, 3, 0, Math.PI*2); ctx.fill();
          });
        };

        drawSkeletonDebug(controller.player.skeleton, '#38bdf8');
        controller.entities.forEach(ent => {
          if (ent.type !== 'crate') {
            drawSkeletonDebug(ent.skeleton, ent.state === 'dead' ? '#64748b' : '#f43f5e');
          } else {
            const scX = (ent.x - controller.cameraX) + viewWidth / 2;
            const scY = ent.y - (ent.height || 50) / 2 + camYOffset;
            ctx.strokeStyle = '#eab308';
            ctx.strokeRect(scX - (ent.width || 50)/2, scY - (ent.height || 50)/2, ent.width || 50, ent.height || 50);
          }
        });
      }

      // 4. Compact Minimal Health Bars
      for (const ent of controller.entities) {
        if (ent.state === 'dead') continue;
        const scX = (ent.x - controller.cameraX) + viewWidth / 2;
        if (scX > 0 && scX < viewWidth) {
          const scY = ent.y - (ent.type === 'crate' ? (ent.height || 50) : 135) + camYOffset;
          const barW = 36;
          ctx.fillStyle = '#1e293b';
          ctx.fillRect(scX - barW/2, scY - 8, barW, 3);
          
          const healthRatio = Math.max(0, ent.health / ent.maxHealth);
          ctx.fillStyle = ent.type === 'crate' ? '#eab308' : '#f43f5e';
          ctx.fillRect(scX - barW/2, scY - 8, barW * healthRatio, 3);
        }
      }

      // 5. Highly Optimized Floating Texts (Framerate friendly)
      for (const ft of controller.floatingTexts) {
        const scX = (ft.x - controller.cameraX) + viewWidth / 2;
        const scY = ft.y + camYOffset;
        
        ctx.save();
        ctx.font = `900 ${Math.floor(18 * ft.scale)}px sans-serif`;
        ctx.textAlign = 'center';
        ctx.fillStyle = ft.color;
        ctx.shadowColor = ft.color;
        ctx.shadowBlur = 6;
        
        const alpha = ft.life > 0.6 ? (1 - ft.life) / 0.4 : 1;
        ctx.globalAlpha = alpha;
        ctx.fillText(ft.text, scX, scY);
        ctx.restore();
      }

      ctx.restore();
      animationId = requestAnimationFrame(render);
    };

    animationId = requestAnimationFrame(render);

    return () => {
      cancelAnimationFrame(animationId);
      window.removeEventListener('resize', handleResize);
    };
  }, [controller, themeColor]);

  return (
    <div className="relative w-full h-full select-none overflow-hidden">
      <canvas ref={canvasRef} className="w-full h-full block cursor-crosshair" />
    </div>
  );
};
