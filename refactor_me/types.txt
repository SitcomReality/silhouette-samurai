export interface Point {
  x: number;
  y: number;
}

export interface Skeleton {
  head: Point;
  headRadius: number;
  torsoTop: Point;
  torsoBottom: Point;
  torsoRadius: number;
  // Left arm
  leftShoulder: Point;
  leftElbow: Point;
  leftHand: Point;
  armRadius: number;
  // Right arm
  rightShoulder: Point;
  rightElbow: Point;
  rightHand: Point;
  // Left leg
  leftHip: Point;
  leftKnee: Point;
  leftFoot: Point;
  legRadius: number;
  // Right leg
  rightHip: Point;
  rightKnee: Point;
  rightFoot: Point;
}

export type AttackType = 'none' | 'kick_low' | 'roundhouse' | 'slam';

export interface Entity {
  id: string;
  type: 'player' | 'enemy' | 'crate';
  x: number;
  y: number; // base position (ground level or airborne)
  vx: number;
  vy: number;
  facing: 'left' | 'right';
  isGrounded: boolean;
  health: number;
  maxHealth: number;
  state: 'idle' | 'run' | 'attack' | 'hit' | 'dead';
  stateTime: number;
  attackType: AttackType;
  attackTimer: number;
  combo: number;
  skeleton: Skeleton;
  // For crates
  width?: number;
  height?: number;
}

export interface Particle {
  id: string;
  x: number;
  y: number;
  vx: number;
  vy: number;
  radius: number;
  life: number;
  maxLife: number;
  isGrounded: boolean;
  shape: 'circle' | 'square' | 'capsule';
  length?: number;
  angle?: number;
  vAngle?: number;
}

export interface Shockwave {
  id: string;
  x: number;
  radius: number;
  speed: number;
  amplitude: number;
  life: number;
  maxLife: number;
}

export interface FloatingText {
  id: string;
  text: string;
  x: number;
  y: number;
  color: string;
  life: number;
  scale: number;
}

export interface GameStats {
  score: number;
  maxCombo: number;
  enemiesDefeated: number;
}
